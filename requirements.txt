# Requirements for PFG_Sourcing.py
# Python 3.14 (project venv). Versions left unpinned so pip resolves
# releases compatible with the interpreter.
#
# Install:
#   pip install -r requirements.txt
#   playwright install chrome   # download the browser Playwright drives

# --- PDF parsing ---
pdfplumber            # PDF text/word extraction — PFG_Sourcing.py (import: pdfplumber)
pymupdf               # PDF engine for PFG_Validation.py gate checks (import: fitz)

# --- Fuzzy matching (validation gates) ---
rapidfuzz             # fuzzy title/name matching in PFG_Validation.py (import: rapidfuzz)

# --- Date parsing (validation FYE checks) ---
python-dateutil       # flexible date parsing (import: dateutil); also a pandas dep

# --- Excel I/O ---
openpyxl              # read/write .xlsm / .xlsx workbooks (import: openpyxl)

# --- Data / logging (parquet log tracker) ---
pandas                # DataFrames, read_parquet / to_parquet (import: pandas)
pyarrow               # parquet engine required by pandas .to_parquet()/.read_parquet()

# --- Database ---
pyodbc                # SQL Server connection via ODBC (import: pyodbc)

# --- Environment config ---
python-dotenv         # load DB credentials from .env (import: dotenv)

# --- Concurrency / cross-process file locking ---
filelock              # cross-process lock around the parquet log (import: filelock)

# --- Browser automation (FAC search + downloads) ---
playwright            # headless Chrome automation (import: playwright)

# --- Imaging (validation proof popups/screenshots) ---
Pillow                # image handling for Tk proof popups (import: PIL)

# NOTE: tkinter is used but ships with the standard CPython installer on
# Windows; it is not installable via pip.

# --- LLM providers (extraction engine: Extraction/pipeline.py via PFG_Extraction.py) ---
openai                # OpenAI SDK (import: openai) — hard top-level import in Extraction/pipeline.py
google-genai          # Gemini SDK (import: google.genai) — default extraction provider
anthropic             # Claude SDK (import: anthropic) — used when provider=claude

# --- Extraction engine extras ---
pypdf                 # PDF page slicing in Extraction/page_extractor.py (import: pypdf)
polars                # parquet log writer for the extraction pipeline (import: polars)
# NOTE: pdfplumber / pdfminer.six / pypdfium2 / numpy are already pulled in above
# (pdfplumber is listed under PDF parsing; the rest arrive transitively).
